cmake_minimum_required(VERSION 3.16)
project(vless_mesh LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(CURL REQUIRED)
add_executable(node-agent src/main.cpp)
target_link_libraries(node-agent PRIVATE yaml-cpp nlohmann_json::nlohmann_json CURL::libcurl)
target_compile_options(node-agent PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS node-agent DESTINATION bin)
